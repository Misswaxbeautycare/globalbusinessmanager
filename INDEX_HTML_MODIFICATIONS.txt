// ========== COMMANDES CLIENT ==========
function afficherCommandes(cat){
  var commandes = DB.get('commandesClient');
  if(cat && cat !== 'tous') commandes = commandes.filter(function(c){ return c.statut === cat; });
  
  var total = commandes.reduce(function(s,c){ return s+c.montantTotal; },0);
  document.getElementById('cmd-total').textContent = fmt(total);
  document.getElementById('cmd-count').textContent = commandes.length;

  var el = document.getElementById('cmd-liste');
  if(!commandes.length){
    el.innerHTML = '<div class="vide"><div class="vide-ico">📥</div><div class="vide-txt">Aucune commande</div><div class="vide-sub">Les commandes apparaîtront ici quand les clients passent des commandes via la boutique</div></div>';
    return;
  }

  el.innerHTML = '<table><thead><tr><th>#</th><th>Date</th><th>Client</th><th>Téléphone</th><th>Pays</th><th>Montant</th><th>Statut</th><th>Actions</th></tr></thead><tbody>'+
    commandes.map(function(c,i){
      var statutBadge = c.statut === 'payee' ? '<span class="b bv">✅ PAYÉE</span>' : '<span class="b bo">⏳ EN ATTENTE</span>';
      return '<tr><td><strong>'+c.id+'</strong></td><td>'+fmtDate(c.date)+'</td><td>'+c.clientNom+'</td><td>'+c.clientTel+'</td><td>'+c.clientPays+'</td><td><strong>'+fmt(c.montantTotal)+'</strong></td><td>'+statutBadge+'</td>'+
        '<td><div style="display:flex;gap:6px">'+
        '<button class="btn bw bsm" onclick="afficherDetailCommande(\''+c.id+'\')">👁️</button>'+
        '<button class="btn bw bsm" onclick="changerStatutCommande(\''+c.id+'\',\''+(c.statut==='en_attente'?'payee':'en_attente')+'\')">🔄</button>'+
        '<button class="btn bw bsm" onclick="supprimerCommande(\''+c.id+'\')">🗑️</button>'+
        '</div></td></tr>';
    }).join('')+'</tbody></table>';
}

function afficherDetailCommande(id){
  var c = DB.get('commandesClient').find(function(x){ return x.id === id; });
  if(!c) return;
  var produits = [];
  try { produits = JSON.parse(c.produits || '[]'); } catch(e) {}
  var html = '<div style="margin-bottom:20px"><div style="display:grid;grid-template-columns:1fr 1fr;gap:16px;margin-bottom:16px"><div><div class="kl">Clients</div><div style="font-size:15px;font-weight:700">'+c.clientNom+'</div></div><div><div class="kl">Téléphone</div><div style="font-size:15px;font-weight:700">'+c.clientTel+'</div></div><div><div class="kl">Email</div><div style="font-size:15px;font-weight:700">'+c.clientEmail+'</div></div><div><div class="kl">Pays</div><div style="font-size:15px;font-weight:700">'+c.clientPays+'</div></div><div><div class="kl">Date</div><div style="font-size:15px;font-weight:700">'+fmtDate(c.date)+'</div></div><div><div class="kl">Statut</div><div style="font-size:15px;font-weight:700">'+(c.statut==='payee'?'✅ PAYÉE':'⏳ EN ATTENTE')+'</div></div></div></div>';
  html += '<div style="background:var(--gris);padding:16px;border-radius:12px;margin-bottom:16px"><div class="kl" style="margin-bottom:12px">📦 Produits</div><table style="width:100%"><thead><tr style="background:var(--fonce2);color:#fff"><th style="padding:10px;text-align:left">Produit</th><th style="padding:10px;text-align:center">Qté</th><th style="padding:10px;text-align:center">Type</th><th style="padding:10px;text-align:right">Sous-total</th></tr></thead><tbody>';
  produits.forEach(function(p){
    html += '<tr style="border-bottom:1px solid var(--bord)"><td style="padding:10px">'+p.nom+'</td><td style="padding:10px;text-align:center">'+p.qte+' '+p.unite+'</td><td style="padding:10px;text-align:center">'+p.typeAchat.toUpperCase()+'</td><td style="padding:10px;text-align:right;font-weight:800">'+fmt(p.total)+'</td></tr>';
  });
  html += '</tbody></table></div><div style="display:flex;gap:10px"><button class="btn bn bsm" onclick="envoyerMsgCommande(\''+c.id+'\')">📲 WhatsApp</button><button class="btn bsm" style="background:var(--wine);color:#fff" onclick="envoyerEmailCommande(\''+c.id+'\')">📧 Email</button><button class="btn bw bsm" onclick="changerStatutCommande(\''+c.id+'\',\''+(c.statut==='en_attente'?'payee':'en_attente')+'\')">🔄 Marquer</button></div>';
  document.getElementById('cmd-detail-cont').innerHTML = html;
  document.getElementById('cmd-detail-zone').style.display = 'block';
}

function fermerDetailCommande(){ document.getElementById('cmd-detail-zone').style.display = 'none'; }
function changerStatutCommande(id, s){ var c = DB.get('commandesClient').find(function(x){ return x.id === id; }); if(!c) return; c.statut = s; CACHE.commandesClient = CACHE.commandesClient.map(function(x){ return x.id === id ? c : x; }); sbUpdate('commandesClient', c); afficherCommandes('tous'); fermerDetailCommande(); alert('✅ Statut mis à jour !'); }
function supprimerCommande(id){ if(!confirm('Supprimer?')) return; CACHE.commandesClient = CACHE.commandesClient.filter(function(c){ return c.id !== id; }); sbDelete('commandesClient', id); rafraichir('commandes'); }
function envoyerMsgCommande(id){ var c = DB.get('commandesClient').find(function(x){ return x.id === id; }); if(!c) return; var msg = 'COMMANDE '+c.id+'\n\nTOTAL: '+fmt(c.montantTotal); window.open('https://wa.me/'+c.clientTel.replace(/[\s\+\-]/g,'')+'?text='+encodeURIComponent(msg), '_blank'); }
function envoyerEmailCommande(id){ var c = DB.get('commandesClient').find(function(x){ return x.id === id; }); if(!c) return; window.open('mailto:'+c.clientEmail+'?subject='+encodeURIComponent('Commande '+c.id)+'&body='+encodeURIComponent('Récapitulatif: '+fmt(c.montantTotal)), '_blank'); }
function filtrerCommandes(btn){ document.querySelectorAll('#pills-commandes .pill').forEach(function(b){ b.classList.remove('on'); }); btn.classList.add('on'); afficherCommandes(btn.dataset.cat); }
function exporterCommandes(){ var commandes = DB.get('commandesClient'); var csv = 'ID,Date,Client,Total,Statut\n'; commandes.forEach(function(c){ csv += c.id+','+c.date+','+c.clientNom+','+c.montantTotal+','+c.statut+'\n'; }); var a = document.createElement('a'); a.href = 'data:text/csv;charset=utf-8,'+encodeURIComponent(csv); a.download = 'commandes.csv'; a.click(); }
